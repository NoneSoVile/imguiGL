function(group_sources_by_folder target)
    file(GLOB_RECURSE SRC_FILES CONFIGURE_DEPENDS
        ${CMAKE_SOURCE_DIR}/main.cpp
        ${CMAKE_SOURCE_DIR}/src/*.cpp
    )
    file(GLOB_RECURSE HDR_FILES CONFIGURE_DEPENDS
        ${CMAKE_SOURCE_DIR}/include/*.h
        ${CMAKE_SOURCE_DIR}/include/*.hpp
    )

    set(ALL_FILES ${SRC_FILES} ${HDR_FILES})

    foreach(FILE_PATH IN LISTS ALL_FILES)    
        file(RELATIVE_PATH REL_PATH "${CMAKE_SOURCE_DIR}" "${FILE_PATH}")

        get_filename_component(FOLDER "${REL_PATH}" PATH)

        string(REPLACE "/" "\\" GROUP "${FOLDER}")

        source_group("${GROUP}" FILES "${FILE_PATH}")

        list(APPEND FINAL_SRC_LIST "${FILE_PATH}")
    endforeach()


    target_sources(${target} PRIVATE ${FINAL_SRC_LIST})
endfunction()
